cmake_minimum_required(VERSION 3.8)
project(guide_robot_journal)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(guide_robot_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(Threads REQUIRED)
add_library(journal_core src/core.cpp src/config.cpp src/store.cpp)
target_include_directories(journal_core PUBLIC include)
target_link_libraries(journal_core PUBLIC yaml-cpp Threads::Threads)
add_library(journal_node src/node.cpp)
target_link_libraries(journal_node PUBLIC journal_core)
ament_target_dependencies(journal_node PUBLIC rclcpp std_msgs guide_robot_msgs sensor_msgs nav_msgs geometry_msgs diagnostic_msgs)
add_executable(robot_journal src/main.cpp)
target_link_libraries(robot_journal journal_node)
install(TARGETS robot_journal DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS tools/extract_cdr.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp)
  target_link_libraries(test_core journal_core)
  ament_add_gtest(test_node test/test_node.cpp TIMEOUT 60)
  target_link_libraries(test_node journal_node)
endif()
ament_package()
